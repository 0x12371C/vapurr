poly_data sync

# Windows (PowerShell) -- needs Python 3.10+, uv, HyperSync token
powershell -c "irm https://astral.sh/uv/install.ps1 | iex"
git clone https://github.com/warproxxx/poly_data.git
cd poly_data
# Create .env with HYPERSYNC_API from https://envio.dev/app/api-tokens
# Token must include HyperSync product access.
uv sync
uv run poly-data
# First run: full markets + chain backfill (hours to days on free tier).
# Later runs: incremental; usually seconds.
# Output: processed/trades.csv (who/when/price/size/counterparty).
